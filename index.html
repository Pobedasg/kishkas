SQL для Supabase

Зайди в SQL Editor і виконай цей запит:

create table if not exists notes (
  id uuid default gen_random_uuid() primary key,
  user_id uuid references auth.users,
  title text, content text,
  created_at timestamptz default now()
);
create table if not exists stats (
  id uuid default gen_random_uuid() primary key,
  user_id uuid references auth.users,
  month text, year int,
  ig int default 0, tt int default 0,
  posts int default 0, stories int default 0,
  reels int default 0, reach int default 0,
  note text, created_at timestamptz default now()
);
alter table notes enable row level security;
alter table stats enable row level security;
create policy if not exists "own_notes" on notes
  using (auth.uid()=user_id) with check (auth.uid()=user_id);
create policy if not exists "own_stats" on stats
  using (auth.uid()=user_id) with check (auth.uid()=user_id);
